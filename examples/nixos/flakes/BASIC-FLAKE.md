# Basic NixOS Flake Structure

A minimal but complete Flake structure for NixOS configurations.

## Use Case

Starting point for any NixOS configuration using Flakes.

## flake.nix

```nix
{
  description = "NixOS configuration for my-hostname";

  inputs = {
    # NixOS stable (always use stable, not unstable)
    nixpkgs.url = "github:NixOS/nixpkgs/nixos-24.05";

    # Optional: Home Manager for user configuration
    # home-manager = {
    #   url = "github:nix-community/home-manager/release-24.05";
    #   inputs.nixpkgs.follows = "nixpkgs";
    # };

    # Optional: agenix for secrets
    # agenix = {
    #   url = "github:ryantm/agenix";
    #   inputs.nixpkgs.follows = "nixpkgs";
    # };
  };

  outputs = { self, nixpkgs, ... }@inputs: {
    nixosConfigurations.my-hostname = nixpkgs.lib.nixosSystem {
      system = "x86_64-linux";

      # Pass inputs to modules
      specialArgs = { inherit inputs; };

      modules = [
        ./configuration.nix
        ./hardware-configuration.nix

        # Optional: Home Manager module
        # inputs.home-manager.nixosModules.home-manager

        # Optional: agenix module
        # inputs.agenix.nixosModules.default
      ];
    };

    # Multiple hosts example
    # nixosConfigurations.other-host = nixpkgs.lib.nixosSystem {
    #   system = "x86_64-linux";
    #   modules = [ ./hosts/other-host/configuration.nix ];
    # };
  };
}
```

## configuration.nix

```nix
{ config, pkgs, inputs, ... }:

{
  # System basics
  system.stateVersion = "24.05";

  # Boot loader (adjust for your system)
  boot.loader.systemd-boot.enable = true;
  boot.loader.efi.canTouchEfiVariables = true;

  # Networking
  networking.hostName = "my-hostname";
  networking.networkmanager.enable = true;

  # Timezone and locale
  time.timeZone = "Europe/London";
  i18n.defaultLocale = "en_GB.UTF-8";

  # Users
  users.users.myuser = {
    isNormalUser = true;
    extraGroups = [ "wheel" "networkmanager" ];
  };

  # Allow unfree packages if needed
  nixpkgs.config.allowUnfree = true;

  # Enable Flakes
  nix.settings.experimental-features = [ "nix-command" "flakes" ];

  # Basic packages
  environment.systemPackages = with pkgs; [
    vim
    git
    curl
    htop
  ];
}
```

## Directory Structure

```
nixos-config/
├── flake.nix              # Flake definition
├── flake.lock             # Locked dependencies (generated)
├── configuration.nix      # Main configuration
├── hardware-configuration.nix  # Hardware (generated by nixos-generate-config)
└── modules/               # Custom modules (optional)
    └── my-service.nix
```

## Commands

```bash
# Check flake syntax
nix flake check

# Update flake inputs
nix flake update

# Build without switching
nixos-rebuild build --flake .#my-hostname

# Switch to new configuration
sudo nixos-rebuild switch --flake .#my-hostname

# Test without persisting
sudo nixos-rebuild test --flake .#my-hostname
```

## Variations

### ARM64 (e.g., Raspberry Pi)

```nix
nixosConfigurations.my-pi = nixpkgs.lib.nixosSystem {
  system = "aarch64-linux";
  modules = [ ./configuration.nix ];
};
```

### Cross-compilation

```nix
nixosConfigurations.my-pi = nixpkgs.lib.nixosSystem {
  system = "aarch64-linux";
  modules = [
    ./configuration.nix
    {
      nixpkgs.buildPlatform = "x86_64-linux";
    }
  ];
};
```
